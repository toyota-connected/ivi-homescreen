# shell/logging/tests/compat-matrix/CMakeLists.txt
#
# Standalone compat-matrix harness for the DLT bridge. A whitebox test: it
# compiles the bridge sources (now in shared/src/logging) directly into the
# compat_smoke executable and exercises the mux header's macros plus the
# bridge's ring/worker path across the C++17/20/23 toolchains.
#
# Used by .github/workflows/ihs-logging-compat-matrix.yml. Does not depend on
# Wayland, Flutter engine, or any homescreen third_party submodule, so it runs
# on a bare image with only a C++ toolchain.

cmake_minimum_required(VERSION 3.16)
project(ihs_logging_compat_matrix CXX)

if(NOT CMAKE_CXX_STANDARD)
    set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS        OFF)

# Locate the ivi-homescreen source root (four levels up from this file).
get_filename_component(IHS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.."
                       ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${IHS_ROOT}/cmake")

option(ENABLE_DLT "Enable DLT logging" ON)
include(cxx_compat)

set(IHS_SHARED_DIR ${IHS_ROOT}/shared)

# The bridge sources include ihs/ihs_export.h, generated by
# generate_export_header when building libihs_shared. This harness compiles the
# sources straight into the test binary, so nothing is actually exported —
# supply an equivalent no-op export header.
set(IHS_GEN_INC ${CMAKE_BINARY_DIR}/ihs_gen)
file(WRITE ${IHS_GEN_INC}/ihs/ihs_export.h
"#ifndef IHS_EXPORT_H\n#define IHS_EXPORT_H\n#define IHS_EXPORT\n#define IHS_NO_EXPORT\n#endif\n")

set(IHS_BRIDGE_SRCS
    ${IHS_SHARED_DIR}/src/logging/libdlt_loader.cpp
    ${IHS_SHARED_DIR}/src/logging/thread_ring.cpp
    ${IHS_SHARED_DIR}/src/logging/ring_registry.cpp
    ${IHS_SHARED_DIR}/src/logging/context_cache.cpp
    ${IHS_SHARED_DIR}/src/logging/worker.cpp
    ${IHS_SHARED_DIR}/src/logging/bridge.cpp
    ${IHS_SHARED_DIR}/src/logging/ffi_shim.cpp
)

add_executable(compat_smoke compat_smoke.cc)

if(ENABLE_DLT)
    target_sources(compat_smoke PRIVATE ${IHS_BRIDGE_SRCS})
    target_include_directories(compat_smoke PRIVATE
        ${IHS_SHARED_DIR}/src/logging
        ${IHS_SHARED_DIR}/include
        ${IHS_GEN_INC}
        ${IHS_ROOT}/shell/logging
    )
    target_compile_definitions(compat_smoke PRIVATE ENABLE_DLT=1)
    find_package(Threads REQUIRED)
    target_link_libraries(compat_smoke PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
    ihs_apply_cxx_compat(compat_smoke)
else()
    target_include_directories(compat_smoke PRIVATE ${IHS_ROOT}/shell/logging)
endif()

target_compile_options(compat_smoke PRIVATE -Wall -Wextra -Werror)

enable_testing()
add_test(NAME compat_smoke COMMAND compat_smoke)
